### 2.2 잠재적 문제 시나리오

**발생 가능 케이스:**
```
1. SCH 파일에서 등록: NP.SUS_CTR_BAS
   -> tables: table_name='SUS_CTR_BAS', table_owner='NP'
   -> components: component_name='SUS_CTR_BAS' (owner 정보 없음)

2. SQL 파싱 시 추출:
   - 케이스 A: NP.SUS_CTR_BAS (점)
   - 케이스 B: NP_SUS_CTR_BAS (언더스코어)

3. 매칭 실패:
   - 'NP_SUS_CTR_BAS'는 components에 없음
   - 새로운 테이블로 인식하여 INFERRED 생성
```

### 2.3 구현 방안

#### 방안 1: 테이블명 정규화 유틸리티 추가 (권장)

**새 파일 생성: util/table_name_normalizer.py**

```python
"""
테이블명 정규화 유틸리티
- 점(.) vs 언더스코어(_) 혼용 처리
- 스키마.테이블 형식 통일
"""

class TableNameNormalizer:
    """테이블명 정규화 클래스"""

    @staticmethod
    def normalize(table_name: str, owner: str = None) -> str:
        """
        테이블명 정규화

        Args:
            table_name: 테이블명 (예: SUS_CTR_BAS, NP.SUS_CTR_BAS, NP_SUS_CTR_BAS)
            owner: 스키마 소유자 (예: NP, SAMPLE)

        Returns:
            정규화된 테이블명 (예: NP.SUS_CTR_BAS)
        """
        if not table_name:
            return table_name

        # 이미 점(.)이 포함된 경우
        if '.' in table_name:
            parts = table_name.split('.', 1)
            schema = parts[0].strip().upper()
            table = parts[1].strip().upper()
            return f"{schema}.{table}"

        # owner가 제공된 경우
        if owner:
            return f"{owner.upper()}.{table_name.upper()}"

        # 단순 테이블명
        return table_name.upper()

    @staticmethod
    def split(table_name: str) -> tuple:
        """
        테이블명을 스키마와 테이블로 분리

        Returns:
            (schema, table_name) 튜플
        """
        if '.' in table_name:
            parts = table_name.split('.', 1)
            return (parts[0].strip().upper(), parts[1].strip().upper())

        # 언더스코어(_) 첫 번째 부분을 스키마로 간주 (선택사항)
        # 예: NP_SUS_CTR_BAS -> (NP, SUS_CTR_BAS)
        # 주의: 이 로직은 프로젝트 명명 규칙에 따라 조정 필요
        return (None, table_name.upper())

    @staticmethod
    def match(table1: str, table2: str) -> bool:
        """
        두 테이블명이 동일한지 비교
        점(.)과 언더스코어(_)를 동일하게 처리

        Examples:
            NP.SUS_CTR_BAS == NP_SUS_CTR_BAS -> True
            SAMPLE.USERS == USERS -> False (스키마 불일치)
        """
        t1_normalized = table1.upper().replace('_', '').replace('.', '')
        t2_normalized = table2.upper().replace('_', '').replace('.', '')
        return t1_normalized == t2_normalized
```

## 3. USE_TABLE 관계 중복 방지

### 3.1 현황 분석

**중복 발생 현황:**
```
finalQuery -> USERS_ (2회, rel_ids: 392, 400)
inventoryQuery -> PRODUCTS_ (2회, rel_ids: 393, 413)
query -> USERS (2회, rel_ids: 103, 199)
queryBuilder -> USERS_ (2회, rel_ids: 391, 399)
sql -> USERS (2회, rel_ids: 210, 220)
```

**중복 원인:**
1. 동일한 SQL 컴포넌트에서 동일한 테이블을 여러 번 참조
2. relationships INSERT 시 중복 체크 없음
3. UNIQUE INDEX가 (src_id, dst_id, rel_type)만 포함하고 있어 실제로는 중복 방지 되어야 하지만, 버그로 중복 INSERT 발생

### 3.2 구현 방안

#### 방안 1: relationships INSERT 전 중복 체크 (권장)

**util/relationship_utils.py 신규 생성:**

```python
"""
relationships 관리 유틸리티
- 중복 방지 INSERT
- 순서 정규화
"""

from util.database_utils import DatabaseUtils
from util.logger import app_logger

class RelationshipManager:
    """relationships 테이블 관리 클래스"""

    def __init__(self, db_utils: DatabaseUtils):
        self.db_utils = db_utils
        self._cache = set()  # 중복 체크 캐시 (메모리 최적화)

    def insert_use_table(self, src_id: int, dst_id: int, project_id: int) -> bool:
        """
        USE_TABLE 관계 INSERT (중복 체크)

        Returns:
            True: INSERT 성공, False: 이미 존재하여 SKIP
        """
        # 캐시 체크
        cache_key = (src_id, dst_id, 'USE_TABLE')
        if cache_key in self._cache:
            return False

        # DB 체크
        query = """
            SELECT relationship_id
            FROM relationships
            WHERE src_id = ? AND dst_id = ? AND rel_type = 'USE_TABLE' AND del_yn = 'N'
            LIMIT 1
        """
        existing = self.db_utils.execute_query(query, (src_id, dst_id))

        if existing:
            app_logger.debug(f"USE_TABLE 중복 SKIP: src_id={src_id}, dst_id={dst_id}")
            self._cache.add(cache_key)
            return False

        # INSERT
        insert_query = """
            INSERT INTO relationships (src_id, dst_id, rel_type, confidence, del_yn)
            VALUES (?, ?, 'USE_TABLE', 1.0, 'N')
        """
        self.db_utils.execute_insert(insert_query, (src_id, dst_id))
        self._cache.add(cache_key)
        app_logger.debug(f"USE_TABLE INSERT: src_id={src_id}, dst_id={dst_id}")
        return True

    def insert_join(self, src_id: int, dst_id: int, rel_type: str,
                   src_column: str, dst_column: str, join_condition: str,
                   confidence: float = 1.0, normalize_direction: bool = True) -> bool:
        """
        JOIN 관계 INSERT (중복 체크 + 순서 정규화)

        Args:
            normalize_direction: True이면 PK-FK 기반으로 방향 정규화

        Returns:
            True: INSERT 성공, False: 이미 존재하여 SKIP
        """
        # 방향 정규화 (ERD 닭발 로직 참고)
        if normalize_direction:
            src_id, dst_id, src_column, dst_column = self._normalize_join_direction(
                src_id, dst_id, src_column, dst_column
            )

        # 중복 체크 (양방향 모두 확인)
        cache_key_forward = (src_id, dst_id, rel_type, join_condition)
        cache_key_backward = (dst_id, src_id, rel_type, self._reverse_condition(join_condition))

        if cache_key_forward in self._cache or cache_key_backward in self._cache:
            return False

        # DB 체크
        query = """
            SELECT relationship_id
            FROM relationships
            WHERE ((src_id = ? AND dst_id = ?) OR (src_id = ? AND dst_id = ?))
              AND rel_type = ?
              AND del_yn = 'N'
            LIMIT 1
        """
        existing = self.db_utils.execute_query(query, (src_id, dst_id, dst_id, src_id, rel_type))

        if existing:
            app_logger.debug(f"{rel_type} 중복 SKIP: {src_id}-{dst_id}")
            self._cache.add(cache_key_forward)
            return False

        # INSERT
        insert_query = """
            INSERT INTO relationships (
                src_id, dst_id, rel_type,
                src_column, dst_column, join_condition,
                confidence, del_yn
            )
            VALUES (?, ?, ?, ?, ?, ?, ?, 'N')
        """
        self.db_utils.execute_insert(insert_query, (
            src_id, dst_id, rel_type,
            src_column, dst_column, join_condition,
            confidence
        ))
        self._cache.add(cache_key_forward)
        app_logger.debug(f"{rel_type} INSERT: {src_id}-{dst_id}")
        return True

    def _normalize_join_direction(self, src_id, dst_id, src_col, dst_col) -> tuple:
        """
        JOIN 관계 방향 정규화 (PK-FK 기반)

        규칙:
          - src: PK 테이블 (1 side)
          - dst: FK 테이블 (N side)

        Returns:
            (정규화된 src_id, dst_id, src_col, dst_col)
        """
        # PK 여부 확인
        src_is_pk = self._is_pk_column(src_id, src_col)
        dst_is_pk = self._is_pk_column(dst_id, dst_col)

        # PK-FK 관계가 명확한 경우
        if src_is_pk and not dst_is_pk:
            # 이미 올바른 방향 (src=PK, dst=FK)
            return (src_id, dst_id, src_col, dst_col)
        elif not src_is_pk and dst_is_pk:
            # 역방향 -> 뒤집기 (dst=PK, src=FK)
            return (dst_id, src_id, dst_col, src_col)
        else:
            # 불명확한 경우: ID 순서로 정렬 (작은 ID가 src)
            if src_id <= dst_id:
                return (src_id, dst_id, src_col, dst_col)
            else:
                return (dst_id, src_id, dst_col, src_col)

    def _is_pk_column(self, component_id: int, column_name: str) -> bool:
        """컬럼이 PK인지 확인"""
        query = """
            SELECT c.position_pk
            FROM components comp
            JOIN tables t ON comp.component_name = t.table_name
            JOIN columns c ON t.table_id = c.table_id
            WHERE comp.component_id = ?
              AND c.column_name = ?
              AND c.position_pk IS NOT NULL
              AND comp.del_yn = 'N'
              AND t.del_yn = 'N'
              AND c.del_yn = 'N'
            LIMIT 1
        """
        result = self.db_utils.execute_query(query, (component_id, column_name))
        return len(result) > 0

    def _reverse_condition(self, condition: str) -> str:
        """조인 조건 역방향 변환 (A=B -> B=A)"""
        if not condition or '=' not in condition:
            return condition
        parts = condition.split('=', 1)
        return f"{parts[1].strip()} = {parts[0].strip()}"
```

### 4.2 ERD 닭발 표시 로직 분석

**erd_report_generator.py:327-340 분석:**

```python
# ERD 관계 방향 결정: PK가 있는 쪽이 1(왼쪽), FK가 있는 쪽이 N(오른쪽)
# Mermaid 문법: A ||--o{ B 는 A(1) : B(N) 관계를 의미

if rel_info['src_is_pk'] and not rel_info['dst_is_pk']:
    # src가 PK, dst가 FK → src(1) : dst(N)
    one_side = src_label
    many_side = dst_label
elif not rel_info['src_is_pk'] and rel_info['dst_is_pk']:
    # src가 FK, dst가 PK → dst(1) : src(N)
    one_side = dst_label
    many_side = src_label
else:
    # PK-FK 관계가 명확하지 않은 경우, 기본적으로 src → dst 방향 사용
    one_side = src_label
    many_side = dst_label

# ERD는 단순 문법만 지원: A ||--o{ B : has 형태만 사용
mermaid_lines.append(f'"{one_side}" ||--o{{ "{many_side}" : {relationship_label}')
```

**핵심 규칙:**
1. **PK 테이블이 1(one) side**
2. **FK 테이블이 N(many) side**
3. **one_side ||--o{ many_side** (PK 테이블 -> FK 테이블)

### 4.3 구현 방안

#### relationships 저장 규칙 통일

**규칙:**
```
relationships 테이블 저장 시:
  - src_id: PK 테이블의 component_id (1 side, one_side)
  - dst_id: FK 테이블의 component_id (N side, many_side)
  - src_column: PK 컬럼
  - dst_column: FK 컬럼
```

**효과:**
1. 양방향 중복 방지 (A->B만 저장, B->A는 저장 안함)
2. ERD 생성 시 추가 로직 불필요
3. 관계 조회 시 일관된 방향

#### RelationshipManager._normalize_join_direction() 세부 로직

```python
def _normalize_join_direction(self, src_id, dst_id, src_col, dst_col) -> tuple:
    """
    JOIN 관계 방향 정규화 (ERD 닭발 로직과 동일)

    우선순위:
      1. PK-FK 관계 확인 (조인 컬럼 기준)
      2. 다른 PK 존재 여부 확인
      3. INFERRED 여부 확인
      4. component_id 크기 비교

    Returns:
        (정규화된 src_id, dst_id, src_col, dst_col)
    """
    src_table_name = self._get_table_name(src_id)
    dst_table_name = self._get_table_name(dst_id)

    # 우선순위 1: 조인 컬럼의 PK 여부
    src_col_is_pk = self._is_pk_column(src_id, src_col)
    dst_col_is_pk = self._is_pk_column(dst_id, dst_col)

    if src_col_is_pk and not dst_col_is_pk:
        # src가 PK, dst가 FK -> 올바른 방향
        return (src_id, dst_id, src_col, dst_col)
    elif not src_col_is_pk and dst_col_is_pk:
        # src가 FK, dst가 PK -> 역방향, 뒤집기
        return (dst_id, src_id, dst_col, src_col)

    # 우선순위 2: 다른 PK 존재 여부
    if not (src_col_is_pk or dst_col_is_pk):
        src_has_other_pk = self._has_other_pk(src_table_name, src_col)
        dst_has_other_pk = self._has_other_pk(dst_table_name, dst_col)

        if src_has_other_pk and not dst_has_other_pk:
            # src 테이블에만 다른 PK 존재 -> dst가 1, src가 N
            return (dst_id, src_id, dst_col, src_col)
        elif not src_has_other_pk and dst_has_other_pk:
            # dst 테이블에만 다른 PK 존재 -> src가 1, dst가 N
            return (src_id, dst_id, src_col, dst_col)

    # 우선순위 3: INFERRED 여부
    src_is_inferred = self._is_inferred(src_id)
    dst_is_inferred = self._is_inferred(dst_id)

    if src_is_inferred and not dst_is_inferred:
        # src가 INFERRED -> src가 1, dst가 N
        return (src_id, dst_id, src_col, dst_col)
    elif not src_is_inferred and dst_is_inferred:
        # dst가 INFERRED -> dst가 1, src가 N
        return (dst_id, src_id, dst_col, src_col)

    # 우선순위 4: ID 크기 비교 (작은 ID가 src)
    if src_id <= dst_id:
        return (src_id, dst_id, src_col, dst_col)
    else:
        return (dst_id, src_id, dst_col, src_col)

def _has_other_pk(self, table_name: str, exclude_column: str) -> bool:
    """테이블에 조인 컬럼이 아닌 다른 PK가 존재하는지 확인"""
    query = """
        SELECT c.column_name
        FROM columns c
        JOIN tables t ON c.table_id = t.table_id
        WHERE t.table_name = ?
          AND c.position_pk IS NOT NULL
          AND c.column_name != ?
          AND t.del_yn = 'N'
          AND c.del_yn = 'N'
        LIMIT 1
    """
    result = self.db_utils.execute_query(query, (table_name.upper(), exclude_column.upper()))
    return len(result) > 0

def _is_inferred(self, component_id: int) -> bool:
    """컴포넌트가 INFERRED인지 확인 (file_type != 'SCH')"""
    query = """
        SELECT f.file_type
        FROM components c
        JOIN files f ON c.file_id = f.file_id
        WHERE c.component_id = ?
          AND c.del_yn = 'N'
          AND f.del_yn = 'N'
    """
    result = self.db_utils.execute_query(query, (component_id,))
    if result:
        # CSV -> SCH 변경 반영
        return result[0]['file_type'] not in ('CSV', 'SCH')
    return False
```

-- 1. 양방향 중복 확인 (결과 0개여야 함)
SELECT
    r1.relationship_id, r2.relationship_id,
    src1.component_name AS src1, dst1.component_name AS dst1
FROM relationships r1
JOIN relationships r2
    ON r1.src_id = r2.dst_id
    AND r1.dst_id = r2.src_id
    AND r1.rel_type = r2.rel_type
    AND r1.relationship_id < r2.relationship_id
JOIN components src1 ON r1.src_id = src1.component_id
JOIN components dst1 ON r1.dst_id = dst1.component_id
WHERE r1.rel_type LIKE 'JOIN_%'
  AND r1.del_yn = 'N'
  AND r2.del_yn = 'N';

-- 2. 방향 정확도 확인
SELECT
    CASE
        WHEN src_pk > 0 AND dst_pk = 0 THEN '올바름'
        WHEN src_pk = 0 AND dst_pk > 0 THEN '역방향'
        ELSE '불명확'
    END AS direction_status,
    COUNT(*) AS count
FROM (
    SELECT
        r.relationship_id,
        (SELECT COUNT(*)
         FROM columns c
         JOIN tables t ON c.table_id = t.table_id
         WHERE t.table_name = src.component_name
           AND c.column_name = r.src_column
           AND c.position_pk IS NOT NULL) AS src_pk,
        (SELECT COUNT(*)
         FROM columns c
         JOIN tables t ON c.table_id = t.table_id
         WHERE t.table_name = dst.component_name
           AND c.column_name = r.dst_column
           AND c.position_pk IS NOT NULL) AS dst_pk
    FROM relationships r
    JOIN components src ON r.src_id = src.component_id
    JOIN components dst ON r.dst_id = dst.component_id
    WHERE r.rel_type LIKE 'JOIN_%'
      AND r.del_yn = 'N'
      AND r.src_column IS NOT NULL
      AND r.dst_column IS NOT NULL
)
GROUP BY direction_status;
