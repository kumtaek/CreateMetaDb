### 2. RelationshipManager 생성 (P1)

#### 2-1. util/relationship_utils.py (신규)

**중요**: 기존 database_utils.py의 upsert 로직을 **재사용**하여 중복 개발 방지

```python
"""
relationships 테이블 관리 유틸리티

이 모듈은 relationships 테이블의 중복 INSERT를 방지하고,
ERD 닭발 표시 로직과 동일한 방향으로 JOIN 관계를 정규화합니다.

주요 기능:
  - USE_TABLE 관계 중복 방지
  - JOIN 관계 양방향 중복 방지
  - PK-FK 기반 JOIN 방향 정규화 (ERD 로직과 동일)
"""

from typing import Optional, Tuple
from util.database_utils import DatabaseUtils
from util.logger import app_logger, handle_error


class RelationshipManager:
    """
    relationships 테이블 관리 클래스

    중복 방지 및 방향 정규화 기능 제공
    """

    def __init__(self, db_utils: DatabaseUtils):
        """
        초기화

        Args:
            db_utils: 데이터베이스 유틸리티 인스턴스
        """
        self.db_utils = db_utils
        self._cache = set()  # 중복 체크 캐시 (성능 최적화)

    def insert_use_table(self, src_id: int, dst_id: int, project_id: int) -> bool:
        """
        USE_TABLE 관계 INSERT (중복 체크)

        Args:
            src_id: 소스 컴포넌트 ID (SQL 컴포넌트)
            dst_id: 대상 컴포넌트 ID (TABLE 컴포넌트)
            project_id: 프로젝트 ID

        Returns:
            bool: INSERT 성공 시 True, 중복으로 SKIP 시 False
        """
        try:
            # 캐시 체크 (성능 최적화)
            cache_key = (src_id, dst_id, 'USE_TABLE')
            if cache_key in self._cache:
                return False

            # DB 중복 체크
            query = """
                SELECT relationship_id
                FROM relationships
                WHERE src_id = ?
                  AND dst_id = ?
                  AND rel_type = 'USE_TABLE'
                  AND del_yn = 'N'
                LIMIT 1
            """
            existing = self.db_utils.execute_query(query, (src_id, dst_id))

            if existing:
                app_logger.debug(f"USE_TABLE 중복 SKIP: src_id={src_id}, dst_id={dst_id}")
                self._cache.add(cache_key)
                return False

            # INSERT (database_utils의 기존 로직 재사용)
            data = {
                'src_id': src_id,
                'dst_id': dst_id,
                'rel_type': 'USE_TABLE',
                'confidence': 1.0,
                'del_yn': 'N'
            }

            # 기존 upsert 로직 재사용 (중복 개발 방지)
            unique_columns = ['src_id', 'dst_id', 'rel_type']
            result = self.db_utils.upsert('relationships', data, unique_columns)

            if result:
                self._cache.add(cache_key)
                app_logger.debug(f"USE_TABLE INSERT 성공: src_id={src_id}, dst_id={dst_id}")
                return True

            return False

        except Exception as e:
            handle_error(e, f"USE_TABLE INSERT 실패: src_id={src_id}, dst_id={dst_id}")
            return False

    def insert_join(self,
                   src_id: int, dst_id: int,
                   rel_type: str,
                   src_column: str, dst_column: str,
                   join_condition: str,
                   confidence: float = 1.0) -> bool:
        """
        JOIN 관계 INSERT (양방향 중복 방지 + 방향 정규화)

        ERD 닭발 표시 로직(erd_report_generator.py:327-397)과 동일한
        PK-FK 기반 방향 정규화를 적용합니다.

        정규화 규칙:
          - src: PK 테이블 (1 side, one_side)
          - dst: FK 테이블 (N side, many_side)

        Args:
            src_id: 소스 테이블 component_id
            dst_id: 대상 테이블 component_id
            rel_type: 관계 타입 (JOIN_EXPLICIT, JOIN_IMPLICIT, JOIN_MERGE)
            src_column: 소스 조인 컬럼
            dst_column: 대상 조인 컬럼
            join_condition: 조인 조건 원문
            confidence: 신뢰도 (기본값 1.0)

        Returns:
            bool: INSERT 성공 시 True, 중복으로 SKIP 시 False
        """
        try:
            # 1. 방향 정규화 (ERD 로직과 동일)
            src_id, dst_id, src_column, dst_column = self._normalize_join_direction(
                src_id, dst_id, src_column, dst_column
            )

            # 2. 양방향 중복 체크 (캐시)
            cache_key_forward = (src_id, dst_id, rel_type, join_condition)
            cache_key_backward = (dst_id, src_id, rel_type, self._reverse_condition(join_condition))

            if cache_key_forward in self._cache or cache_key_backward in self._cache:
                return False

            # 3. DB 양방향 중복 체크
            query = """
                SELECT relationship_id
                FROM relationships
                WHERE ((src_id = ? AND dst_id = ?) OR (src_id = ? AND dst_id = ?))
                  AND rel_type = ?
                  AND del_yn = 'N'
                LIMIT 1
            """
            existing = self.db_utils.execute_query(
                query, (src_id, dst_id, dst_id, src_id, rel_type)
            )

            if existing:
                app_logger.debug(f"{rel_type} 중복 SKIP: {src_id}-{dst_id}")
                self._cache.add(cache_key_forward)
                return False

            # 4. INSERT (기존 upsert 로직 재사용)
            data = {
                'src_id': src_id,
                'dst_id': dst_id,
                'rel_type': rel_type,
                'src_column': src_column,
                'dst_column': dst_column,
                'join_condition': join_condition,
                'confidence': confidence,
                'del_yn': 'N'
            }

            unique_columns = ['src_id', 'dst_id', 'rel_type']
            result = self.db_utils.upsert('relationships', data, unique_columns)

            if result:
                self._cache.add(cache_key_forward)
                app_logger.debug(f"{rel_type} INSERT 성공: {src_id}-{dst_id}")
                return True

            return False

        except Exception as e:
            handle_error(e, f"{rel_type} INSERT 실패: {src_id}-{dst_id}")
            return False

    def _normalize_join_direction(self,
                                  src_id: int, dst_id: int,
                                  src_col: str, dst_col: str) -> Tuple[int, int, str, str]:
        """
        JOIN 관계 방향 정규화 (ERD 닭발 로직과 동일)

        출처: erd_report_generator.py:355-397 (_get_relationship_info)

        우선순위:
          1. 조인 컬럼의 PK 여부 확인
          2. 테이블의 다른 PK 존재 여부 확인
          3. INFERRED 여부 확인
          4. component_id 크기 비교 (일관성 유지)

        Args:
            src_id: 소스 테이블 component_id
            dst_id: 대상 테이블 component_id
            src_col: 소스 조인 컬럼
            dst_col: 대상 조인 컬럼

        Returns:
            Tuple[int, int, str, str]: (정규화된 src_id, dst_id, src_col, dst_col)
        """
        try:
            # 우선순위 1: 조인 컬럼의 PK 여부
            src_col_is_pk = self._is_pk_column(src_id, src_col)
            dst_col_is_pk = self._is_pk_column(dst_id, dst_col)

            if src_col_is_pk and not dst_col_is_pk:
                # src=PK, dst=FK → 올바른 방향
                return (src_id, dst_id, src_col, dst_col)
            elif not src_col_is_pk and dst_col_is_pk:
                # src=FK, dst=PK → 역방향, 뒤집기
                return (dst_id, src_id, dst_col, src_col)

            # 우선순위 2: 다른 PK 존재 여부
            if not (src_col_is_pk or dst_col_is_pk):
                src_table_name = self._get_table_name(src_id)
                dst_table_name = self._get_table_name(dst_id)

                src_has_other_pk = self._has_other_pk(src_table_name, src_col)
                dst_has_other_pk = self._has_other_pk(dst_table_name, dst_col)

                if src_has_other_pk and not dst_has_other_pk:
                    # src 테이블에만 다른 PK 존재 → dst(1), src(N)
                    return (dst_id, src_id, dst_col, src_col)
                elif not src_has_other_pk and dst_has_other_pk:
                    # dst 테이블에만 다른 PK 존재 → src(1), dst(N)
                    return (src_id, dst_id, src_col, dst_col)

            # 우선순위 3: INFERRED 여부
            src_is_inferred = self._is_inferred(src_id)
            dst_is_inferred = self._is_inferred(dst_id)

            if src_is_inferred and not dst_is_inferred:
                # src가 INFERRED → src(1), dst(N)
                return (src_id, dst_id, src_col, dst_col)
            elif not src_is_inferred and dst_is_inferred:
                # dst가 INFERRED → dst(1), src(N)
                return (dst_id, src_id, dst_col, src_col)

            # 우선순위 4: ID 크기 비교 (일관성 유지)
            if src_id <= dst_id:
                return (src_id, dst_id, src_col, dst_col)
            else:
                return (dst_id, src_id, dst_col, src_col)

        except Exception as e:
            app_logger.error(f"방향 정규화 실패, 원본 유지: {e}")
            # 실패 시 원본 유지
            return (src_id, dst_id, src_col, dst_col)

    def _is_pk_column(self, component_id: int, column_name: str) -> bool:
        """
        컬럼이 PK인지 확인

        Args:
            component_id: 테이블 component_id
            column_name: 확인할 컬럼명

        Returns:
            bool: PK이면 True
        """
        try:
            if not column_name:
                return False

            query = """
                SELECT c.position_pk
                FROM components comp
                JOIN tables t ON comp.component_name = t.table_name
                JOIN columns c ON t.table_id = c.table_id
                WHERE comp.component_id = ?
                  AND c.column_name = ?
                  AND c.position_pk IS NOT NULL
                  AND comp.del_yn = 'N'
                  AND t.del_yn = 'N'
                  AND c.del_yn = 'N'
                LIMIT 1
            """
            result = self.db_utils.execute_query(query, (component_id, column_name.upper()))
            return len(result) > 0

        except Exception as e:
            app_logger.error(f"PK 확인 실패: component_id={component_id}, column={column_name}, {e}")
            return False

    def _has_other_pk(self, table_name: str, exclude_column: str) -> bool:
        """
        테이블에 조인 컬럼이 아닌 다른 PK가 존재하는지 확인

        출처: erd_report_generator.py:399-419 (_has_other_pk)

        Args:
            table_name: 테이블명
            exclude_column: 제외할 컬럼명 (조인 컬럼)

        Returns:
            bool: 다른 PK가 있으면 True
        """
        try:
            if not table_name:
                return False

            query = """
                SELECT c.column_name
                FROM columns c
                JOIN tables t ON c.table_id = t.table_id
                WHERE t.table_name = ?
                  AND c.position_pk IS NOT NULL
                  AND c.column_name != ?
                  AND t.del_yn = 'N'
                  AND c.del_yn = 'N'
                LIMIT 1
            """
            result = self.db_utils.execute_query(
                query, (table_name.upper(), exclude_column.upper())
            )
            return len(result) > 0

        except Exception as e:
            app_logger.error(f"다른 PK 확인 실패: table={table_name}, {e}")
            return False

    def _is_inferred(self, component_id: int) -> bool:
        """
        컴포넌트가 INFERRED인지 확인 (file_type이 CSV/SCH가 아님)

        Args:
            component_id: 확인할 component_id

        Returns:
            bool: INFERRED이면 True
        """
        try:
            query = """
                SELECT f.file_type
                FROM components c
                JOIN files f ON c.file_id = f.file_id
                WHERE c.component_id = ?
                  AND c.del_yn = 'N'
                  AND f.del_yn = 'N'
                LIMIT 1
            """
            result = self.db_utils.execute_query(query, (component_id,))

            if result:
                file_type = result[0]['file_type']
                # CSV/SCH 호환성 (하드코딩 금지 원칙)
                return file_type not in ('CSV', 'SCH')

            return False

        except Exception as e:
            app_logger.error(f"INFERRED 확인 실패: component_id={component_id}, {e}")
            return False

    def _get_table_name(self, component_id: int) -> str:
        """
        component_id로 테이블명 조회

        Args:
            component_id: 테이블 component_id

        Returns:
            str: 테이블명 (조회 실패 시 빈 문자열)
        """
        try:
            query = """
                SELECT component_name
                FROM components
                WHERE component_id = ?
                  AND component_type = 'TABLE'
                  AND del_yn = 'N'
                LIMIT 1
            """
            result = self.db_utils.execute_query(query, (component_id,))
            return result[0]['component_name'] if result else ''

        except Exception as e:
            app_logger.error(f"테이블명 조회 실패: component_id={component_id}, {e}")
            return ''

    def _reverse_condition(self, condition: str) -> str:
        """
        조인 조건 역방향 변환 (A=B -> B=A)

        Args:
            condition: 원본 조인 조건

        Returns:
            str: 역방향 조인 조건
        """
        try:
            if not condition or '=' not in condition:
                return condition

            parts = condition.split('=', 1)
            if len(parts) != 2:
                return condition

            return f"{parts[1].strip()} = {parts[0].strip()}"

        except Exception as e:
            app_logger.error(f"조인 조건 역방향 변환 실패: {e}")
            return condition
```

```sql
-- 1. USE_TABLE 중복 확인 (0건)
SELECT src.component_name AS sql_comp,
       dst.component_name AS table_name,
       COUNT(*) AS dup_count
FROM relationships r
JOIN components src ON r.src_id = src.component_id
JOIN components dst ON r.dst_id = dst.component_id
WHERE r.rel_type = 'USE_TABLE' AND r.del_yn = 'N'
GROUP BY src.component_name, dst.component_name
HAVING COUNT(*) > 1;

-- 2. JOIN 양방향 중복 확인 (0건)
SELECT r1.relationship_id, r2.relationship_id,
       src1.component_name AS r1_src,
       dst1.component_name AS r1_dst
FROM relationships r1
JOIN relationships r2
  ON r1.src_id = r2.dst_id
  AND r1.dst_id = r2.src_id
  AND r1.rel_type = r2.rel_type
  AND r1.relationship_id < r2.relationship_id
JOIN components src1 ON r1.src_id = src1.component_id
JOIN components dst1 ON r1.dst_id = dst1.component_id
WHERE r1.rel_type LIKE 'JOIN_%'
  AND r1.del_yn = 'N'
  AND r2.del_yn = 'N';

-- 3. JOIN 방향 정확도 (올바름 >= 80%)
SELECT
    CASE
        WHEN src_pk > 0 AND dst_pk = 0 THEN '올바름(src=PK)'
        WHEN src_pk = 0 AND dst_pk > 0 THEN '역방향(dst=PK)'
        ELSE '불명확'
    END AS direction_status,
    COUNT(*) AS count
FROM (
    SELECT
        (SELECT COUNT(*) FROM columns c
         JOIN tables t ON c.table_id = t.table_id
         WHERE t.table_name = src.component_name
           AND c.column_name = r.src_column
           AND c.position_pk IS NOT NULL
           AND t.del_yn = 'N' AND c.del_yn = 'N') AS src_pk,
        (SELECT COUNT(*) FROM columns c
         JOIN tables t ON c.table_id = t.table_id
         WHERE t.table_name = dst.component_name
           AND c.column_name = r.dst_column
           AND c.position_pk IS NOT NULL
           AND t.del_yn = 'N' AND c.del_yn = 'N') AS dst_pk
    FROM relationships r
    JOIN components src ON r.src_id = src.component_id
    JOIN components dst ON r.dst_id = dst.component_id
    WHERE r.rel_type LIKE 'JOIN_%'
      AND r.del_yn = 'N'
      AND r.src_column IS NOT NULL
      AND r.dst_column IS NOT NULL
)
GROUP BY direction_status;
```
