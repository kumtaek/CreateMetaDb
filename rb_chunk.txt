    def execute_db_relationship_backfill(project_name: str, conn: sqlite3.Connection) -> Dict[str, int]:
        """Create relationships from existing DB data (backfill).

        - JSP → API_URL: link JSP components to API_URL as CALL_API
        - API_URL → METHOD: link by controller map or best-effort fallback as CALL_METHOD
        """
    stats = {"CALL_API": 0, "CALL_METHOD": 0, "CALL_QUERY": 0, "USE_TABLE": 0}
    try:
        cur = conn.cursor()

        # 1) JSP ??API_URL (CALL_API)
        q1 = (
            "SELECT api.component_id AS api_id, jsp.component_id AS jsp_id "
            "FROM components api "
            "JOIN files f ON api.file_id = f.file_id "
            "JOIN components jsp ON jsp.file_id = f.file_id "
            "WHERE api.component_type = 'API_URL' "
            "AND jsp.component_type = 'JSP' "
            "AND api.del_yn = 'N' AND jsp.del_yn = 'N'"
        )
        for api_id, jsp_id in cur.execute(q1).fetchall():
            try:
                _insert_rel_raw(conn, jsp_id, api_id, 'CALL_API')
                stats["CALL_API"] += 1
            except Exception:
                pass
