```python
    """컴포넌트 유효성 검사"""
    if component.type.startswith('SQL_'):
        # 테이블 존재 여부 확인
        for table in component.referenced_tables:
            if table not in schema_info.tables:
                component.has_error = True
                component.error_message = f"Table {table} does not exist"
                
    elif component.type == 'METHOD':
        # Java 구문 검사 (기본적인 중괄호, 세미콜론 검사)
```

#### 2. 중복 제거 시스템
```python
def normalize_and_deduplicate(components):
    """중복 제거 및 정규화"""
    normalized = {}
    for comp in components:
        # 파일 경로 정규화
        normalized_path = comp.file_path.replace('\\', '/').replace('//', '/')
        
        # 고유키 생성
        unique_key = (comp.name, comp.type, normalized_path)
        
        if unique_key not in normalized:
            normalized[unique_key] = comp
            
    return list(normalized.values())
```

```python
def extract_api_urls_from_jsp(jsp_content):
    """JSP에서 API URL 추출 강화"""
    api_urls = []
    
    # Ajax 패턴
    ajax_patterns = [
        r'\.ajax\s*\(\s*[\'"]([^\'\"]+)[\'"]',
        r'\.post\s*\(\s*[\'"]([^\'\"]+)[\'"]',
        r'\.get\s*\(\s*[\'"]([^\'\"]+)[\'"]',
        r'fetch\s*\(\s*[\'"]([^\'\"]+)[\'"]'
    ]
    
    # Form action 패턴
    form_pattern = r'<form[^>]+action\s*=\s*[\'"]([^\'\"]+)[\'"]'
    
    for pattern in ajax_patterns + [form_pattern]:
        matches = re.findall(pattern, jsp_content, re.IGNORECASE)
        api_urls.extend(matches)
        
    return api_urls
```
